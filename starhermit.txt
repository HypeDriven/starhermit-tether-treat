# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com by hash; never edit it). Re-run `npm run package`
# after every change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.30.0
engine.minor=32
engine.capabilities=platform,storage,clipboard

# Keyboard actions (players rebind keys and pads in the runtime's Options).
# The room is also played with the pointer or a finger: swipe across a rope
# (or tap it) to snip it, tap a bubble holding the treat to pop it, tap a fan
# to switch it; the action list beside the room is a button per legal action.
engine.group.prev=Targets
control.prev=ArrowLeft+ArrowUp | Previous target
control.next=ArrowRight+ArrowDown | Next target
control.confirm=Enter+Space | Confirm focused action
engine.group.pause=Round
control.pause=Escape | Pause
control.undo=KeyU | Undo
control.hint=KeyH | Hint
control.restart=KeyR | Restart
control.camera=KeyC | Reset camera

# Gamepad (the browser build's input.js): the D-pad or left stick cycles the
# targets, South (A) confirms, East (B) and Start pause, North (Y) hints;
# West undoes, the left shoulder restarts, the right stick press resets the camera.
engine.gamepad.prev=DpadLeft+DpadUp+LeftStickLeft+LeftStickUp
engine.gamepad.next=DpadRight+DpadDown+LeftStickRight+LeftStickDown
engine.gamepad.confirm=South
engine.gamepad.pause=East+Start
engine.gamepad.hint=North
engine.gamepad.undo=West
engine.gamepad.restart=LeftShoulder
engine.gamepad.camera=RightStickPress
